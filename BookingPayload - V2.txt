openapi: 3.0.1
info:
  title: AFH Booking API
  version: 1.0.0
  description: >
    Booking API for LeadTech-driven scheduling.
    Flow: Availability -> Hold -> Confirm (or Cancel).

servers:
  - url: https://localhost:7149/api
    description: Local

paths:
  /v1/availability:
    post:
      summary: Get ranked adviser availability slots
      description: >
        Submit client context and proposed start/duration. Returns advisers with slots and a score (1-5)
        indicating best fit.
      operationId: GetAvailability
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: "#/components/schemas/GetAvailabilityRequest"
            examples:
              example:
                value:
                  transactionId: "LT-977788878"
                  clientId: "00QP300000S9sE"
                  proposedStartUtc: "2025-12-18T09:00:00Z"
                  durationMinutes: 60
                  meetingMode: "Remote"
                  meetingType: "Review"
                  page:
                    limit: 10
                    cursor: null
      responses:
        "200":
          description: Ranked availability
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/GetAvailabilityResponse"
              examples:
                example:
                  value:
                    transactionId: "LT-977788878"
                    clientId: "00QP300000S9sE"
                    proposedStartUtc: "2025-12-18T09:00:00Z"
                    durationMinutes: 60
                    meetingMode: "Remote"
                    meetingType: "Review"
                    page:
                      limit: 10
                      cursor: null
                      nextCursor: "eyJvZmZzZXQiOjEwfQ=="
                    advisers:
                      - adviserId: "ADV-001"
                        adviserName: "Isaac Botchway"
                        slots:
                          - slotId: "slot_ADV-001_20251218T090000Z_60"
                            startUtc: "2025-12-18T09:00:00Z"
                            endUtc: "2025-12-18T10:00:00Z"
                            score: 5
                            scoreReason:
                              - "No calendar conflicts"
                              - "Matches meeting mode"
                      - adviserId: "ADV-014"
                        adviserName: "A. Example"
                        slots:
                          - slotId: "slot_ADV-014_20251218T093000Z_60"
                            startUtc: "2025-12-18T09:30:00Z"
                            endUtc: "2025-12-18T10:30:00Z"
                            score: 3
                            scoreReason:
                              - "Minor travel/spacing penalty"
        "400":
          description: Invalid request
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"

  /v1/bookings/hold:
    post:
      summary: Create a booking hold from a chosen slot
      description: >
        Creates a short-lived reservation to prevent double-booking. The hold expires automatically.
      operationId: CreateHold
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: "#/components/schemas/CreateHoldRequest"
            examples:
              example:
                value:
                  transactionId: "LT-977788878"
                  clientId: "00QP300000S9sE"
                  slotId: "slot_ADV-001_20251218T090000Z_60"
                  adviserId: "ADV-001"
                  holdDuration: "00:05:00"
                  subject: "Initial consultation"
                  notes: "Created via LeadTech"
      responses:
        "201":
          description: Hold created
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/CreateHoldResponse"
              examples:
                example:
                  value:
                    bookingId: "bkg_9f2f1c3a8e9b4f6c"
                    transactionId: "LT-977788878"
                    clientId: "00QP300000S9sE"
                    adviserId: "ADV-001"
                    adviserName: "Isaac Botchway"
                    slotId: "slot_ADV-001_20251218T090000Z_60"
                    startUtc: "2025-12-18T09:00:00Z"
                    endUtc: "2025-12-18T10:00:00Z"
                    status: "Hold"
                    holdExpiresUtc: "2025-12-18T09:05:00Z"
        "400":
          description: Invalid request
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"
        "409":
          description: Slot already held/confirmed or conflict detected
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"

  /v1/bookings/{bookingId}/confirm:
    post:
      summary: Confirm a held booking
      description: >
        Finalises the booking. Fails if the hold has expired or conflicts exist.
      operationId: ConfirmBooking
      parameters:
        - name: bookingId
          in: path
          required: true
          schema: { type: string }
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: "#/components/schemas/ConfirmBookingRequest"
            examples:
              example:
                value:
                  transactionId: "LT-977788878"
                  clientId: "00QP300000S9sE"
      responses:
        "200":
          description: Booking confirmed
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ConfirmBookingResponse"
              examples:
                example:
                  value:
                    bookingId: "bkg_9f2f1c3a8e9b4f6c"
                    status: "Confirmed"
                    confirmedUtc: "2025-12-18T08:59:30Z"
        "404":
          description: Booking not found
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"
        "409":
          description: Hold expired or conflict detected
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"

  /v1/bookings/{bookingId}/cancel:
    post:
      summary: Cancel a booking (held or confirmed)
      description: >
        Cancels the booking. If held, it releases the slot. If confirmed, it cancels the booking.
      operationId: CancelBooking
      parameters:
        - name: bookingId
          in: path
          required: true
          schema: { type: string }
      requestBody:
        required: false
        content:
          application/json:
            schema:
              $ref: "#/components/schemas/CancelBookingRequest"
            examples:
              example:
                value:
                  reason: "Client requested reschedule"
                  transactionId: "LT-977788878"
                  clientId: "00QP300000S9sE"
      responses:
        "200":
          description: Booking cancelled
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/CancelBookingResponse"
        "404":
          description: Booking not found
          content:
            application/json:
              schema:
                $ref: "#/components/schemas/ProblemDetails"

components:
  schemas:
    GetAvailabilityRequest:
      type: object
      required: [proposedStartUtc, durationMinutes, meetingMode, meetingType]
      properties:
        transactionId:
          type: string
          description: LeadTech transaction id (preferred if present)
        clientId:
          type: string
          description: LeadTech client id (preferred if present)
        proposedStartUtc:
          type: string
          format: date-time
        durationMinutes:
          type: integer
          minimum: 5
          maximum: 480
        meetingMode:
          $ref: "#/components/schemas/MeetingMode"
        meetingType:
          type: string
          description: Business meeting type/category (e.g. Review, NewBusiness)
        page:
          $ref: "#/components/schemas/PageRequest"

    GetAvailabilityResponse:
      type: object
      required: [advisers, page]
      properties:
        transactionId: { type: string, nullable: true }
        clientId: { type: string, nullable: true }
        proposedStartUtc: { type: string, format: date-time }
        durationMinutes: { type: integer }
        meetingMode: { $ref: "#/components/schemas/MeetingMode" }
        meetingType: { type: string }
        page:
          $ref: "#/components/schemas/PageResult"
        advisers:
          type: array
          items:
            $ref: "#/components/schemas/AdviserSlots"

    AdviserSlots:
      type: object
      required: [adviserId, adviserName, slots]
      properties:
        adviserId:
          type: string
          description: Internal adviser id (NOT email)
        adviserName:
          type: string
        slots:
          type: array
          items:
            $ref: "#/components/schemas/Slot"

    Slot:
      type: object
      required: [slotId, startUtc, endUtc, score]
      properties:
        slotId:
          type: string
          description: Stable slot identifier for follow-up hold requests
        startUtc: { type: string, format: date-time }
        endUtc: { type: string, format: date-time }
        score:
          type: integer
          minimum: 1
          maximum: 5
          description: >
            Best-fit rating where 5 is best and 1 is lowest acceptable.
            Intended meaning:
            5 = perfect fit (no conflicts, ideal spacing/rules satisfied)
            4 = strong fit (minor penalties)
            3 = acceptable (some penalties)
            2 = weak fit (significant penalties but still valid)
            1 = last-resort (valid but least desirable)
        scoreReason:
          type: array
          items: { type: string }
          description: Human-readable explanation of score factors

    CreateHoldRequest:
      type: object
      required: [slotId, adviserId, holdDuration]
      properties:
        transactionId: { type: string, nullable: true }
        clientId: { type: string, nullable: true }
        slotId: { type: string }
        adviserId: { type: string }
        holdDuration:
          type: string
          description: Hold duration as TimeSpan (HH:mm:ss)
          example: "00:05:00"
        subject: { type: string, nullable: true }
        notes: { type: string, nullable: true }

    CreateHoldResponse:
      type: object
      required: [bookingId, status, holdExpiresUtc]
      properties:
        bookingId: { type: string }
        transactionId: { type: string, nullable: true }
        clientId: { type: string, nullable: true }
        adviserId: { type: string }
        adviserName: { type: string }
        slotId: { type: string }
        startUtc: { type: string, format: date-time }
        endUtc: { type: string, format: date-time }
        status:
          type: string
          enum: [Hold]
        holdExpiresUtc:
          type: string
          format: date-time

    ConfirmBookingRequest:
      type: object
      properties:
        transactionId: { type: string, nullable: true }
        clientId: { type: string, nullable: true }

    ConfirmBookingResponse:
      type: object
      required: [bookingId, status, confirmedUtc]
      properties:
        bookingId: { type: string }
        status:
          type: string
          enum: [Confirmed]
        confirmedUtc:
          type: string
          format: date-time

    CancelBookingRequest:
      type: object
      properties:
        transactionId: { type: string, nullable: true }
        clientId: { type: string, nullable: true }
        reason: { type: string, nullable: true }

    CancelBookingResponse:
      type: object
      required: [bookingId, status]
      properties:
        bookingId: { type: string }
        status:
          type: string
          enum: [Cancelled]
        cancelledUtc:
          type: string
          format: date-time
          nullable: true

    PageRequest:
      type: object
      properties:
        limit:
          type: integer
          minimum: 1
          maximum: 100
          default: 10
        cursor:
          type: string
          nullable: true
          description: Opaque cursor from a previous response

    PageResult:
      type: object
      required: [limit]
      properties:
        limit: { type: integer }
        cursor: { type: string, nullable: true }
        nextCursor: { type: string, nullable: true }

    MeetingMode:
      type: string
      enum: [Remote, InPerson, Hybrid]

    ProblemDetails:
      type: object
      properties:
        error:
          type: object
          properties:
            code: { type: string }
            message: { type: string }
